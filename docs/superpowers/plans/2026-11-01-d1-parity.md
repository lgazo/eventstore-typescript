> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the D1 store's parity gaps with Postgres: tenancy, `minSequenceNumber`, millisecond timestamps, and prove behavior with a real-workerD1 e2e test suite — all while keeping the single-statement conditional append.

**Architecture:** Same two-layer predicate semantics (SQL `json_extract` pre-filter + JS subset post-filter). Tenant scope AND-ed into query WHERE and append context max check, mirroring ESTS-5 postgres decisions. `occurred_at` bound explicitly as JS ISO string instead of relying on SQLite `CURRENT_TIMESTAMP`. E2E via `miniflare` (real workerd D1, in-process).

**Tech Stack:** TypeScript, jest + ts-jest, node:sqlite shim for unit tests, miniflare (workerd) for e2e.

**Spec:** `docs/superpowers/specs/2026-11-01-d1-parity-design.md`

```ts
export function buildTenantScopeSql(tenantId?: string): { sql: string; params: unknown[] } {
  if (tenantId === undefined) return { sql: 'tenant_id IS NULL', params: [] };
  return { sql: 'tenant_id = ?', params: [tenantId] };
}

export function buildContextQuerySql(query?: EventQuery, tenantId?: string): SqlQuery {
  const tenant = buildTenantScopeSql(tenantId);
  const { where, params: filterParams } = buildContextWhere(query);
  const clauses = [tenant.sql, `(${where})`];
  const params: unknown[] = [...tenant.params, ...filterParams];
  if (query?.options?.minSequenceNumber !== undefined) {
    clauses.push('sequence_number > ?');
    params.push(query.options.minSequenceNumber);
  }
  return {
    sql: `SELECT sequence_number, occurred_at, event_type, payload FROM events WHERE ${clauses.join(' AND ')} ORDER BY sequence_number ASC`,
    params,
  };
}
```

```ts
const tenantClause = buildTenantScopeSql(this.tenantId);
const { where, params: filterParams } = buildContextWhere(query);
const contextParams = [...tenantClause.params, ...filterParams];
const contextWhere = `${tenantClause.sql} AND (${where})`;
const values = events.map(() => '(?, ?)').join(', ');
const occurredAt = new Date().toISOString();
const tenantValue = this.tenantId ?? null;
const sql = `INSERT INTO events (occurred_at, event_type, payload, tenant_id)
  SELECT i.occurred_at, v.column1, json(v.column2), i.tenant_id FROM (VALUES ${values}) AS v,
  (SELECT ? AS occurred_at, ? AS tenant_id) AS i
  WHERE (SELECT COALESCE(MAX(sequence_number), 0) FROM events WHERE ${contextWhere}) = ?
  RETURNING sequence_number, occurred_at, event_type, payload`;
const rows = await this.executeAll<EventRow>(sql, [
  ...events.flatMap((e) => [e.eventType, JSON.stringify(e.payload)]),
  occurredAt, tenantValue, ...contextParams, expectedSequenceNumber,
```
